package copilot
import("context";"testing";"time";"runtime")
func TestReviewEventsBeforeSendReply(t *testing.T) {
 h,_,_:=testHarness(t,"review-event-burst")
 ctx,cancel:=context.WithTimeout(t.Context(),time.Second);defer cancel()
 n:=0
 _,err:=h.Complete(ctx,testRequest(),func(s string)error{n+=len(s);return nil})
 t.Logf("small answer: bytes=%d err=%v",n,err)
 if err!=nil||n!=256{t.Fatalf("pre-reply events prevented protocol progress: bytes=%d err=%v",n,err)}
}
func TestReviewCatalogCancellationRace(t *testing.T) {
 h,_,_:=testHarness(t,"catalog-hold")
 for i:=0;i<100;i++ {
  ctx,cancel:=context.WithTimeout(t.Context(),time.Second)
  client,_,stop,err:=h.connect(ctx)
  if err!=nil{cancel();t.Fatal(err)}
  done:=make(chan struct{})
  go func(){_,_=client.ListModels(ctx);close(done)}()
  runtime.Gosched()
  cancel()
  <-done
  _=stop()
 }
}
